only = set(sys.argv[1:])
for table, rad, cols in jobs:
    if only and table not in only:
        continue
    try:
        t = irsa_cone(table, rad)
    except Exception as e:
        print(f"### {table} r={rad}\": ERROR {e}")
        continue
    print(f"### {table} r={rad}\": {len(t)} rows")
    if len(t) == 0:
        continue
    low = {c.lower(): c for c in t.colnames}
    show = ["sep_arcsec"] + ([low[c.lower()] for c in cols if c.lower() in low] if cols else [c for c in t.colnames if c not in ("clon","clat")][:20])
    t[show][:40].pprint(max_lines=60, max_width=260)
