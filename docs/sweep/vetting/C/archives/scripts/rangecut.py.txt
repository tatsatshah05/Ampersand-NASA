"""Cut a small piece out of a large uncompressed remote FITS image with HTTP byte-range requests
(IRSA's /data/ URLs ignore cutout parameters, but honour Range). Usage: rangecut.py URL OUT.fits [half_arcsec]"""
import sys, math, time, io, warnings
warnings.filterwarnings("ignore")
import numpy as np
import requests
from astropy.io import fits
from astropy.wcs import WCS

UA = {"User-Agent": "Mozilla/5.0 (generic research client)"}
RA, DEC = 241.709120, -52.205661


def rng(url, a, b, tries=8):
    for i in range(tries):
        try:
            r = requests.get(url, headers={**UA, "Range": f"bytes={a}-{b}"}, timeout=120)
            if r.status_code == 206:
                return r.content
            last = r.status_code
        except Exception as e:
            last = repr(e)
        time.sleep(4 + 4 * i)
    raise RuntimeError(f"range request failed: {last}")


def main(url, out, half=60.0):
    # header
    raw = b""
    blk = 0
    while True:
        chunk = rng(url, blk * 2880 * 10, (blk + 1) * 2880 * 10 - 1)
        raw += chunk
        cards = [raw[i:i + 80] for i in range(0, len(raw) - len(raw) % 80, 80)]
        endi = next((i for i, c in enumerate(cards) if c.startswith(b"END     ") or c.strip() == b"END"), None)
        if endi is not None:
            break
        blk += 1
        if blk > 20:
            raise RuntimeError("no END card")
    hlen = int(math.ceil((endi + 1) * 80 / 2880.0) * 2880)
    hdr = fits.Header.fromstring(raw[:(endi + 1) * 80].decode("ascii", "replace"))
    nx, ny, bp = hdr["NAXIS1"], hdr["NAXIS2"], hdr["BITPIX"]
    nb = abs(bp) // 8
    w = WCS(hdr).celestial
    from astropy.coordinates import SkyCoord
    import astropy.units as u
    from astropy.wcs.utils import skycoord_to_pixel
    x, y = (float(v) for v in skycoord_to_pixel(SkyCoord(RA * u.deg, DEC * u.deg), w, origin=0, mode='all'))
    sc = abs(hdr.get("CDELT2", hdr.get("CD2_2", 0)) or 0) * 3600 or float(np.sqrt(abs(np.linalg.det(w.pixel_scale_matrix)))) * 3600
    hp = int(math.ceil(half / sc))
    x0, x1 = max(0, int(x) - hp), min(nx, int(x) + hp + 1)
    y0, y1 = max(0, int(y) - hp), min(ny, int(y) + hp + 1)
    if x0 >= x1 or y0 >= y1:
        raise RuntimeError(f"position outside image ({x:.1f},{y:.1f}) of {nx}x{ny}")
    a = hlen + (y0 * nx) * nb
    b = hlen + (y1 * nx) * nb - 1
    data = rng(url, a, b)
    dt = {8: ">u1", 16: ">i2", 32: ">i4", -32: ">f4", -64: ">f8"}[bp]
    arr = np.frombuffer(data, dtype=dt).reshape(y1 - y0, nx)[:, x0:x1].astype(np.float32)
    if "BSCALE" in hdr or "BZERO" in hdr:
        arr = arr * hdr.get("BSCALE", 1.0) + hdr.get("BZERO", 0.0)
    nh = hdr.copy()
    for k in ("NAXIS1", "NAXIS2", "BSCALE", "BZERO"):
        nh.remove(k, ignore_missing=True)
    nh["CRPIX1"] = hdr["CRPIX1"] - x0
    nh["CRPIX2"] = hdr["CRPIX2"] - y0
    nh["HISTORY"] = f"cut by HTTP range from {url} x {x0}:{x1} y {y0}:{y1}"
    fits.PrimaryHDU(arr, nh).writeto(out, overwrite=True)
    keys = [k for k in hdr if any(s in k for s in ("DATE", "MJD", "OBS", "BUNIT", "AORKEY"))]
    print(out, arr.shape, f"pixel {sc:.2f}\"", {k: hdr[k] for k in keys[:12]})


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2], float(sys.argv[3]) if len(sys.argv) > 3 else 60.0)
