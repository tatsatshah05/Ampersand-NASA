import requests, sys
url = sys.argv[1]
def run(adql):
    r = requests.post(url, data={'REQUEST':'doQuery','LANG':'ADQL','FORMAT':'csv','QUERY':adql}, headers={'User-Agent':'generic research client'}, timeout=120)
    print('--', url, r.status_code, '|', adql); print(r.text[:1500] if r.ok else r.text[:200])
run("SELECT b.main_id, b.ra, b.dec, b.otype, b.nbref, i.id FROM ident AS i JOIN basic AS b ON i.oidref=b.oid WHERE i.id IN ('V* V462 Lup','V* V7994 Sgr','NOVA Lup 2025','NOVA Sgr 2025b','NOVA Sgr 2025','V* V572 Vel','NOVA Vel 2025')")
run("SELECT main_id, ra, dec, otype, nbref FROM basic WHERE CONTAINS(POINT('ICRS',ra,dec), CIRCLE('ICRS',227.01362,-40.14308,0.05))=1 AND otype IN ('No*','No?','CV*')")
run("SELECT TOP 5 main_id, ra, dec, otype FROM basic WHERE main_id LIKE 'V* V7%Sgr' ORDER BY oid DESC")
