"""Read every ATel page in a number range (one request at a time), extract title, date, subjects and every coordinate;
report any telegram with a position within 1 deg of the target or of the control novae, or with keywords."""
import re, sys, time, os, requests, datetime
from astropy.coordinates import SkyCoord
import astropy.units as u
import numpy as np
lo, hi, outdir, summ = int(sys.argv[1]), int(sys.argv[2]), sys.argv[3], sys.argv[4]
P = {'TARGET': SkyCoord(241.70931, -52.20484, unit='deg'), 'V462Lup': SkyCoord(227.01362, -40.14308, unit='deg'), 'V7994Sgr': SkyCoord(270.96983, -31.45739, unit='deg')}
KW = ['Norma', ' Nor ', 'Nor)', 'SPHEREx', 'infrared transient', 'WINTER', 'PGIR', 'Gattini', 'PRIME', 'VVV', 'G330.', '330.4', 'Lupus', 'young stellar', 'YSO', 'FU Ori', 'FUor', 'EXor']
sexa = re.compile(r'(\d{1,2})[h: ]\s?(\d{1,2})[m: ]\s?(\d{1,2}(?:\.\d+)?)s?\s*,?\s*(?:Dec(?:l)?\.?\s*[=:]?\s*)?([+\-\u2212\u2013])\s?(\d{1,2})[d\u00b0: ]\s?(\d{1,2})[m\'\u2032: ]\s?(\d{1,2}(?:\.\d+)?)')
deci = re.compile(r'R\.?A\.?\s*(?:\(J2000\))?\s*[=:]?\s*(\d{1,3}\.\d{2,})\s*(?:deg)?\s*,?\s*(?:and\s*)?Dec(?:l)?\.?\s*(?:\(J2000\))?\s*[=:]?\s*([+\-\u2212]?\d{1,2}\.\d{2,})')
out = open(summ, 'a')
s = requests.Session(); s.headers['User-Agent'] = 'Mozilla/5.0 (generic research client)'
for n in range(lo, hi + 1):
    f = os.path.join(outdir, f'{n}.html')
    if not os.path.exists(f):
        try:
            r = s.get(f'https://www.astronomerstelegram.org/?read={n}', timeout=60)
            open(f, 'wb').write(r.content)
        except Exception as e:
            out.write(f'{n}\tERR\t{e}\n'); continue
        time.sleep(0.6)
    h = open(f, encoding='utf-8', errors='replace').read()
    tm = re.search(r'<H1 class="title">\s*(.*?)</H1>', h, re.S | re.I)
    title = re.sub(r'\s+', ' ', re.sub('<[^>]+>', '', tm.group(1))).strip() if tm else '(no title found)'
    txt = re.sub(r'\s+', ' ', re.sub('<[^>]+>', ' ', h)).replace('&deg;', '\u00b0').replace('&minus;', '-')
    dm = re.search(r'on (\d{1,2} \w{3} \d{4}; \d\d:\d\d UT)', txt)
    sm = re.search(r'Subjects:\s*(.*?)\s*(?:Referred to by|Tweet|Share|$)', txt)
    coords = []
    for m in sexa.finditer(txt):
        g = m.groups(); sign = '-' if g[3] in '-\u2212\u2013' else '+'
        try:
            if int(g[0]) < 24 and int(g[4]) <= 90: coords.append(SkyCoord(f'{g[0]}h{g[1]}m{g[2]}s {sign}{g[4]}d{g[5]}m{g[6]}s'))
        except Exception: pass
    for m in deci.finditer(txt):
        try: coords.append(SkyCoord(float(m.group(1)), float(m.group(2).replace('\u2212', '-')), unit='deg'))
        except Exception: pass
    seps = {k: (min(c.separation(v).deg for c in coords) if coords else np.nan) for k, v in P.items()}
    kws = [k for k in KW if k in txt]
    line = f"{n}\t{dm.group(1) if dm else ''}\t{len(coords)}\t{seps['TARGET']:.3f}\t{seps['V462Lup']:.3f}\t{seps['V7994Sgr']:.3f}\t{'|'.join(kws)}\t{title[:150]}"
    out.write(line + '\n'); out.flush()
