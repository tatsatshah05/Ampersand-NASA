"""Control for the emission-line channels: do field stars in the SAME exposures show a jump too?

For each exposure listed, measure C (at the refined position) and the reference 2MASS stars with
the same aperture (1.5 px, sky ring 5-9 px). Compare each star's flux with its flux in the
neighbouring exposures of the same sequence (the exposures just before and after, same detector).
A calibration fault at a channel would lift every star; a real line lifts only C.
"""
import glob
import json
import math
import sys

import numpy as np
from astropy.io import fits
from astropy.stats import sigma_clipped_stats
from astropy.wcs import WCS
from astropy.wcs.utils import proj_plane_pixel_area
from astropy.time import Time

sys.path.insert(0, sys.argv[1])
from phot_c import coverage, PHOT_BAD, SKY_BAD  # noqa: E402

CUT = "<repository>/backend/cache/sweep/cutouts/4ae14c5ec1"
RA_C, DEC_C = 241.709051, -52.205612        # refined position (median of 48 fits, this work)
refs = json.load(open(sys.argv[2]))


def phot(f, ra, dec, rap=1.5):
    with fits.open(f) as h:
        hd = h["IMAGE"].header
        img = h["IMAGE"].data.astype(float)
        fl = h["FLAGS"].data.astype(np.int64)
        var = h["VARIANCE"].data.astype(float)
        zodi = h["ZODI"].data.astype(float)
        wave = h["WAVE"].data
    w = WCS(hd)
    x, y = (float(v) for v in w.all_world2pix(ra, dec, 0))
    ny, nx = img.shape
    if not (10 <= x < nx - 10 and 10 <= y < ny - 10):
        return None, None
    data = img - zodi
    finite = np.isfinite(data) & np.isfinite(var)
    yy, xx = np.mgrid[0:ny, 0:nx]
    r = np.hypot(xx - x, yy - y)
    ring = (r >= 5) & (r < 9) & finite & ((fl & SKY_BAD) == 0)
    if ring.sum() < 30:
        return None, None
    _, sky, _ = sigma_clipped_stats(data[ring], sigma=3.0, maxiters=5)
    cov = coverage(ny, nx, x, y, rap)
    used = cov > 0
    bad = used & (((fl & PHOT_BAD) != 0) | ~finite)
    if cov[bad].sum() / cov[used].sum() > 0.2:
        return None, float(wave[int(round(y)), int(round(x))])
    good = used & ~bad
    scale = cov[used].sum() / cov[good].sum()
    to_mjy = float(proj_plane_pixel_area(w)) * (math.pi / 180) ** 2 * 1e9
    return float(np.sum(cov[good] * (data[good] - sky)) * scale * to_mjy), float(wave[int(round(y)), int(round(x))])


files = {}
for f in glob.glob(CUT + "/*.fits"):
    h = fits.getheader(f, "IMAGE")
    files[f] = (float(h["MJD-AVG"]), int(h["DETECTOR"]))
# sequences of the line exposures: (date-time prefix, detector)
targets = [("2026-02-27T01:3", 5), ("2026-03-03T00:", 6), ("2026-03-03T00:", 3), ("2026-02-19T02:5", 4),
           ("2026-02-22T19:0", 4), ("2026-02-22T14:", 4), ("2026-02-24T23:0", 5), ("2026-08-03T04:", 6),
           ("2026-08-08T02:", 6), ("2026-08-13T22:1", 5), ("2026-02-27T01:3", 2), ("2026-02-28T13:", 6),
           ("2026-02-28T13:", 3)]
for prefix, det in targets:
    seq = sorted([(m, f) for f, (m, d) in files.items() if d == det and Time(m, format="mjd").isot.startswith(prefix)])
    if not seq:
        continue
    print(f"\n== D{det} sequence {prefix}*  ({len(seq)} exposures)")
    table = []
    for m, f in seq:
        c, um = phot(f, RA_C, DEC_C)
        stars = [phot(f, s["ra"], s["dec"])[0] for s in refs]
        table.append((Time(m, format="mjd").isot[:16], um, c, stars))
    smed = []
    for k in range(len(refs)):
        vals = [t[3][k] for t in table if t[3][k] is not None and t[3][k] > 5]
        smed.append(np.median(vals) if len(vals) >= 2 else None)
    for t, um, c, stars in table:
        ratios = [s / sm for s, sm in zip(stars, smed) if s is not None and sm]
        cs = f"{c:7.1f}" if c is not None else "   None"
        us = f"{um:.3f}" if um is not None else "  -  "
        print(f"{t} {us} C={cs} mJy | ref stars flux/their median in this sequence: n={len(ratios)} "
              f"median {np.median(ratios) if ratios else float('nan'):.3f} range "
              f"{(min(ratios) if ratios else float('nan')):.2f}-{(max(ratios) if ratios else float('nan')):.2f}")
