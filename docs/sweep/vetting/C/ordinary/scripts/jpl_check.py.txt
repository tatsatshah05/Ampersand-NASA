with a control: Horizons gives (2) Pallas's geocentric position at a date, and sb_ident asked at that
position with the same field size must list Pallas."""

SB = "https://ssd-api.jpl.nasa.gov/sb_ident.api"
HZ = "https://ssd.jpl.nasa.gov/api/horizons.api"
RA, DEC = 241.709051, -52.205612


def hms(deg):
    t = (deg % 360) / 15
    h = int(t); m = int((t - h) * 60)
    return f"{h:02d}-{m:02d}-{(t - h - m / 60) * 3600:05.2f}"


def dms(deg):
    s = "M" if deg < 0 else ""
    a = abs(deg); d = int(a); m = int((a - d) * 60)
    return f"{s}{d:02d}-{m:02d}-{(a - d - m / 60) * 3600:05.2f}"


def sb_ident(iso, ra, dec, half=0.05):
    import math
    p = {"mpc-code": "500", "obs-time": iso.replace("T", "_")[:19], "fov-ra-center": hms(ra), "fov-dec-center": dms(dec),
         "fov-ra-hwidth": f"{half / math.cos(math.radians(dec)):.4f}", "fov-dec-hwidth": f"{half:.4f}",
         "two-pass": "true", "suppress-first-pass": "true"}
    for k in range(4):
        try:
            r = S.get(SB, params=p, timeout=180)
            if r.status_code == 200:
                return p, r.json()
            err = f"HTTP {r.status_code}: {r.text[:200]}"
        except Exception as e:  # noqa: BLE001
            err = f"{type(e).__name__}: {e}"
        time.sleep(10 * (k + 1))
    return p, {"error": err}

from concurrent.futures import ThreadPoolExecutor
def one(a):
    lab, iso = a
    p, rep = sb_ident(iso, RA, DEC)
    rows = rep.get("data_second_pass", []) if isinstance(rep, dict) else []
    print(f"{lab}: n={rep.get('n_second_pass')} warning={rep.get('warning')} error={rep.get('error')} rows={rows[:5]}")
    return {"what": lab, "params": p, "reply": rep}
with ThreadPoolExecutor(2) as ex:
    results += list(ex.map(one, asks))
