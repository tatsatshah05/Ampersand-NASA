import re, sys
import numpy as np
from astropy.coordinates import SkyCoord
txt = open(sys.argv[1], errors="replace").read()
trs = re.split(r"<tr[^>]*>", txt, flags=re.I)
rows = []
for tr in trs:
    tds = [re.sub(r"<[^>]+>", "", t).strip() for t in re.findall(r"<td[^>]*>(.*?)</td>", tr, flags=re.S | re.I)]
    co = [i for i, t in enumerate(tds) if re.fullmatch(r"\d{1,2}:\d{1,2}:[\d.]+", t)]
    if not co or co[0] + 2 >= len(tds) or not re.fullmatch(r"[+-]?\d{1,2}:\d{1,2}:[\d.]+", tds[co[0] + 1]):
        continue
    i = co[0]
    rows.append((tds[0], tds[1], tds[i], tds[i + 1], tds[i + 2], tds[-1]))
print("rows parsed:", len(rows), "latest date:", max(r[4] for r in rows))
def todeg(r, d):
    h, m, s = [float(x) for x in r.split(":")]
    sg = -1 if d.strip().startswith("-") else 1
    D, M, S = [abs(float(x)) for x in d.split(":")]
    return (h + m / 60 + s / 3600) * 15, sg * (D + M / 60 + S / 3600)
pos = np.array([todeg(r[2], r[3]) for r in rows])
cat = SkyCoord(pos[:, 0], pos[:, 1], unit="deg")
for tag, (ra, dec) in {"B": (253.18044, -44.67687), "V462Lup": (227.01362, -40.14308), "V7994Sgr": (270.96983, -31.45739)}.items():
    s = SkyCoord(ra, dec, unit="deg").separation(cat).arcsec
    print(tag)
    for i in np.argsort(s)[:3]:
        print(f"   {s[i]/60:7.2f}' {rows[i][0]} | {rows[i][1]} | {rows[i][4]} | {rows[i][5][:110]}")
