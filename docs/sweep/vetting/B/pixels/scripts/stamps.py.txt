"""PNG stamps: per-visit median stacks (north up, east left, 2 arcsec grid, same grey scale per detector) for D6, D4, D2,
and a strip of single D6 exposures. Own resampling (bilinear), archive WCS corrected by the measured shift."""
import json, os, sys
import numpy as np
from PIL import Image, ImageDraw
sys.path.insert(0, os.path.dirname(__file__))
from common import CUT, PHOT_BAD, W, Cut, saturated, offset_en

P = {p["file"]: p for p in json.load(open(f"{W}/out/per_exposure.json"))}
SH = {r["file"]: tuple(r["wcs_shift_px"]) for r in json.load(open(f"{W}/out/pass3.json"))}
AD = json.load(open(f"{W}/out/adopted_positions.json"))
RA_B, DE_B = AD["bright"]
GRID = 2.0      # arcsec per stamp pixel
HALF = 45       # 90 x 90 grid pixels = 3 arcmin
MARKS = {"bright place": AD["bright"], "2MASS J16524373 (Ks 13.8)": [253.182237, -44.676834],
         "41 mJy star 2MASS J16524412": [253.183849, -44.675522], "GLIMPSE G341.0882 ([3.6] 14.2)": [253.179017, -44.676589]}


def grid_sky():
    j, i = np.mgrid[-HALF:HALF, -HALF:HALF]
    east = -(i + 0.5) * GRID          # east to the left
    north = (j + 0.5) * GRID
    north = north[::-1]               # row 0 at the top = north
    east = east[:, :]
    dec = DE_B + north / 3600.0
    ra = RA_B + east / 3600.0 / np.cos(np.radians(DE_B))
    return ra, dec


GRA, GDE = grid_sky()


def resample(fn):
    c = Cut(os.path.join(CUT, fn))
    c.shift = SH.get(fn, (0.0, 0.0))
    d = c.data.copy()
    bad = ((c.flags & PHOT_BAD) != 0) & ~saturated(c.flags, d)   # keep saturated cores visible (they hold numbers)
    d[bad] = np.nan
    v = d[np.isfinite(d)]
    med = np.median(v)
    d = d - med
    x, y = c.sky2pix(GRA.ravel(), GRA.ravel() * 0 + GDE.ravel())
    x = x.reshape(GRA.shape); y = y.reshape(GRA.shape)
    x0 = np.floor(x).astype(int); y0 = np.floor(y).astype(int)
    fx, fy = x - x0, y - y0
    ok = (x0 >= 0) & (y0 >= 0) & (x0 < c.nx - 1) & (y0 < c.ny - 1)
    out = np.full(GRA.shape, np.nan)
    xs, ys = np.clip(x0, 0, c.nx - 2), np.clip(y0, 0, c.ny - 2)
    val = (d[ys, xs] * (1 - fx) * (1 - fy) + d[ys, xs + 1] * fx * (1 - fy) + d[ys + 1, xs] * (1 - fx) * fy + d[ys + 1, xs + 1] * fx * fy)
    out[ok] = val[ok]
    return out


def to_png(img, lo, hi, scale=3):
    a = np.arcsinh((img - lo) / (0.05 * (hi - lo)))
    a0 = np.arcsinh(0.0); a1 = np.arcsinh((hi - lo) / (0.05 * (hi - lo)))
    g = np.clip((a - a0) / (a1 - a0), 0, 1)
    rgb = np.dstack([g, g, g]) * 255
    nan = ~np.isfinite(img)
    rgb[nan] = (40, 60, 160)
    im = Image.fromarray(rgb.astype("uint8")).resize((img.shape[1] * scale, img.shape[0] * scale), Image.NEAREST)
    return im


def mark(im, scale, color=(255, 60, 60)):
    """Short ticks 8-14 arcsec from each marked place (never covering it)."""
    dr = ImageDraw.Draw(im)
    cols = [(255, 70, 70), (70, 200, 255), (255, 200, 0), (120, 255, 120)]
    for (lab, (ra, de)), col in zip(MARKS.items(), cols):
        e, n = offset_en(ra, de, RA_B, DE_B)
        cx = (HALF - e / GRID) * scale; cy = (HALF - n / GRID) * scale
        r1, r2 = 6 / GRID * scale, 11 / GRID * scale
        dr.line([(cx - r2, cy), (cx - r1, cy)], fill=col, width=2)
        dr.line([(cx, cy - r2), (cx, cy - r1)], fill=col, width=2)
    return im


os.makedirs(f"{W}/pictures", exist_ok=True)
SC = 3
tiles = {}
for det in (6, 4, 2):
    stacks = {}
    for ep in (0, 1, 2):
        fns = [f for f, p in P.items() if p["det"] == det and p["ep"] == ep]
        arr = np.array([resample(f) for f in fns])
        with np.errstate(all="ignore"):
            stacks[ep] = (np.nanmedian(arr, axis=0), len(fns))
    allv = np.concatenate([s[0][np.isfinite(s[0])] for s in stacks.values()])
    lo = np.percentile(allv, 5); hi = np.percentile(allv, 99.9)
    for ep in (0, 1, 2):
        tiles[(det, ep)] = (stacks[ep][0], lo, hi, stacks[ep][1])
labels = {0: "Aug-Sep 2025", 1: "Feb-Mar 2026", 2: "Aug 2026"}
waves = {6: "D6 4.4-5.0 um", 4: "D4 2.4-3.8 um", 2: "D2 1.1-1.65 um"}
for marked in (False, True):
    W_, H_ = 3 * 90 * SC + 4 * 10, 3 * (90 * SC + 30) + 40
    canvas = Image.new("RGB", (W_, H_), (255, 255, 255))
    dr = ImageDraw.Draw(canvas)
    dr.text((10, 5), "Candidate B, SPHEREx per-visit median stacks, 3' x 3', north up, east left; same grey scale within a row" + ("; ticks mark catalogue places" if marked else ""), fill=(0, 0, 0))
    for r, det in enumerate((6, 4, 2)):
        for col, ep in enumerate((0, 1, 2)):
            img, lo, hi, n = tiles[(det, ep)]
            im = to_png(img, lo, hi, SC)
            if marked:
                im = mark(im, SC)
            x = 10 + col * (90 * SC + 10); y = 25 + r * (90 * SC + 30)
            canvas.paste(im, (x, y + 15))
            dr.text((x, y), f"{waves[det]}  {labels[ep]}  ({n} exposures)", fill=(0, 0, 0))
    canvas.save(f"{W}/pictures/stamps_visits{'_marked' if marked else ''}.png")
print("visit stamps written")

# strip of single D6 exposures, 1.4 arcmin, same scale
for det in (6, 3):
    rows = []
    for ep in (0, 1, 2):
        fns = sorted([f for f, p in P.items() if p["det"] == det and p["ep"] == ep], key=lambda f: P[f]["mjd"])
        rows.append([(f, resample(f)[HALF - 21:HALF + 21, HALF - 21:HALF + 21]) for f in fns])
    allv = np.concatenate([a[np.isfinite(a)] for row in rows for _, a in row])
    lo, hi = np.percentile(allv, 5), np.percentile(allv, 99.95)
    ncol = max(len(r) for r in rows)
    s2 = 3
    canvas = Image.new("RGB", (ncol * (42 * s2 + 4) + 10, 3 * (42 * s2 + 30) + 30), (255, 255, 255))
    dr = ImageDraw.Draw(canvas)
    dr.text((5, 3), f"Candidate B, every D{det} exposure, 84\" x 84\" centred on the bright place, north up, one grey scale; label: date, wavelength, my forced flux", fill=(0, 0, 0))
    for r, row in enumerate(rows):
        for k, (f, a) in enumerate(row):
            im = to_png(a, lo, hi, s2)
            x = 5 + k * (42 * s2 + 4); y = 20 + r * (42 * s2 + 30)
            canvas.paste(im, (x, y + 22))
            p = P[f]
            dr.text((x, y), f"{p['date'][2:10]} {p['wave']:.2f}", fill=(0, 0, 0))
            dr.text((x, y + 10), "flagged" if p["forced"] is None else f"{p['forced']:.0f} mJy", fill=(0, 0, 0))
    canvas.save(f"{W}/pictures/strip_D{det}_exposures.png")
print("strips written")
