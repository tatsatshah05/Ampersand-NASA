def fit(img, flg, x, y):
    bad = (flg & photometry.PHOTOMETRY_BAD) != 0
    yy, xx = np.mgrid[0:img.shape[0], 0:img.shape[1]]
    ix, iy = int(round(x)), int(round(y))
    ring = (np.hypot(xx-x, yy-y) > 8) & (np.hypot(xx-x, yy-y) < 14) & ~bad & np.isfinite(img)
    sky = sigma_clipped_stats(img[ring])[1]
    box = (np.abs(xx-ix) <= 2) & (np.abs(yy-iy) <= 2) & ~bad & np.isfinite(img)
    if box.sum() < 20: return None
    z = img[box] - sky
    f = lambda p: p[0]*np.exp(-((xx[box]-p[1])**2+(yy[box]-p[2])**2)/(2*p[3]**2)) + p[4] - z
    try:
        r = least_squares(f, [max(z.max(), 1e-3), x, y, 0.6, 0], bounds=([0, x-1.5, y-1.5, 0.3, -np.inf], [np.inf, x+1.5, y+1.5, 2, np.inf]))
    except Exception: return None
    return r.x
