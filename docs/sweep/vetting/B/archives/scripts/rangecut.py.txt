"""Cut a small region out of a large remote uncompressed FITS image with HTTP Range requests.
usage: rangecut.py URL out.fits ra dec half_arcsec"""
import sys, requests, numpy as np, warnings
from astropy.io import fits
from astropy.wcs import WCS
warnings.filterwarnings("ignore")
UA = {"User-Agent": "generic-astro-client/1.0"}
def get(url, a, b):
    r = requests.get(url, headers={**UA, "Range": f"bytes={a}-{b}"}, timeout=180)
    assert r.status_code == 206, (r.status_code, len(r.content))
    return r.content
def cut(url, out, ra, dec, half):
    raw = b""; off = 0
    while True:   # read header blocks until END
        raw += get(url, off, off + 2880*4 - 1); off += 2880*4
        if b"END     " in raw.replace(b"\n", b"") and any(raw[i:i+8] == b"END     " for i in range(0, len(raw), 80)):
            break
    endi = next(i for i in range(0, len(raw), 80) if raw[i:i+8] == b"END     ")
    hlen = ((endi + 80 + 2879) // 2880) * 2880
    hdr = fits.Header.fromstring(raw[:endi+80].decode("ascii"))
    bitpix, nx, ny = hdr["BITPIX"], hdr["NAXIS1"], hdr["NAXIS2"]
    nb = abs(bitpix) // 8
    w = WCS(hdr)
    lon, lat = ra, dec
    if str(hdr.get("CTYPE1", "")).startswith("GLON"):
        from astropy.coordinates import SkyCoord
        g = SkyCoord(ra, dec, unit="deg").galactic; lon, lat = g.l.deg, g.b.deg
    x, y = w.all_world2pix(lon, lat, 0)
    from astropy.wcs.utils import proj_plane_pixel_scales
    pix = float(np.mean(proj_plane_pixel_scales(w.celestial))) * 3600
    h = int(np.ceil(half / pix))
    x0, x1 = max(0, int(x) - h), min(nx, int(x) + h + 1)
    y0, y1 = max(0, int(y) - h), min(ny, int(y) + h + 1)
    if x0 >= x1 or y0 >= y1: raise SystemExit(f"position outside image ({x},{y}) of {nx}x{ny}")
    a = hlen + (y0 * nx + x0) * nb
    b = hlen + ((y1 - 1) * nx + x1) * nb - 1
    if (b - a) > 60e6: raise SystemExit("too big")
    buf = get(url, a, b)
    dt = {8: ">u1", 16: ">i2", 32: ">i4", -32: ">f4", -64: ">f8"}[bitpix]
    arr = np.frombuffer(buf, dtype=dt)
    rows = []
    for j in range(y1 - y0):
        s = j * nx
        rows.append(arr[s: s + (x1 - x0)])
    data = np.array(rows, dtype=float)
    if "BSCALE" in hdr or "BZERO" in hdr:
        data = data * hdr.get("BSCALE", 1.0) + hdr.get("BZERO", 0.0)
    nh = w[y0:y1, x0:x1].to_header()
    for k in ("BUNIT", "DATE_OBS", "DATE-OBS", "MJD_OBS", "MJD-OBS", "TELESCOP", "INSTRUME", "WAVELENG", "FILTER", "BAND"):
        if k in hdr: nh[k] = hdr[k]
    fits.PrimaryHDU(data.astype("float32"), header=nh).writeto(out, overwrite=True)
    print(out, data.shape, f"{pix:.3f}\"/px", f"bytes {b-a+1}")
if __name__ == "__main__":
    cut(sys.argv[1], sys.argv[2], float(sys.argv[3]), float(sys.argv[4]), float(sys.argv[5]))
