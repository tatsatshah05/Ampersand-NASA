"""Blackbody (+ optional foreground extinction) fits to B's spectra per epoch, and luminosities at 3, 8, 12 kpc.
Two inputs: (1) the sweep's per-detector medians (its aperture fluxes, quoted from its tile file); (2) this check's
deblended, field-star-calibrated per-exposure fluxes (out/spectra_points.json) without saturated exposures, without
the two channels that hold Pa-alpha / Br-alpha, and without the CO2-ice channels 4.24-4.30 um.
Extinction law: A_lambda/A_K at 1.25, 1.65, 2.16, 3.6, 4.5, 5.8 um = 2.5, 1.55, 1.0, 0.56, 0.43, 0.43 (Indebetouw+ 2005), log-interpolated.
"""
import json, math, numpy as np
from scipy.optimize import least_squares
h, c, k = 6.62607e-34, 2.99792e8, 1.380649e-23
LAW_L = np.array([1.25, 1.65, 2.16, 3.6, 4.5, 5.8]); LAW_A = np.array([2.5, 1.55, 1.0, 0.56, 0.43, 0.43])
def alaw(lam): return np.exp(np.interp(np.log(lam), np.log(LAW_L), np.log(LAW_A)))
def bnu(lam_um, T):
    nu = c / (lam_um * 1e-6)
    return 2 * h * nu**3 / c**2 / np.expm1(np.minimum(h * nu / (k * T), 700))
def model(p, lam, ak_fixed=None):
    lnom, T = p[0], p[1]; ak = p[2] if ak_fixed is None else ak_fixed
    return np.exp(lnom) * bnu(lam, T) * 1e29 * 10 ** (-0.4 * ak * alaw(lam))   # mJy
def fit(lam, f, e, ak=None, free_ak=False):
    lam, f, e = map(np.asarray, (lam, f, e))
    def res(p):
        return (model(p, lam, None if free_ak else (ak or 0.0)) - f) / e
    best = None
    for T0 in (300, 600, 900, 1500, 3000):
        p0 = [0.0, T0] + ([2.0] if free_ak else [])
        # scale initial omega
        m0 = model(p0, lam, None if free_ak else (ak or 0.0)); p0[0] = math.log(np.median(f / m0))
        lo = [-80, 150] + ([0] if free_ak else []); hi = [20, 30000] + ([15] if free_ak else [])
        r = least_squares(res, p0, bounds=(lo, hi))
        if best is None or r.cost < best.cost: best = r
    chi2 = 2 * best.cost; dof = max(len(f) - len(best.x), 1)
    try:
        cov = np.linalg.inv(best.jac.T @ best.jac) * max(chi2 / dof, 1); err = np.sqrt(np.diag(cov))
    except Exception:
        err = [np.nan] * len(best.x)
    return best.x, err, chi2 / dof
def lum(p, ak, d_kpc):
    # bolometric flux of the unreddened blackbody: sigma T^4 * Omega / pi
    omega = math.exp(p[0]); T = p[1]
    fbol = 5.670374e-8 * T**4 * omega / math.pi   # W/m^2
    d = d_kpc * 3.0857e19
    return 4 * math.pi * d * d * fbol / 3.828e26, fbol
DET_UM = {1: 0.93, 2: 1.38, 3: 2.03, 4: 3.12, 5: 4.12, 6: 4.71}
out = {}
# (1) the sweep's medians (aperture, includes the neighbour 10" away)
sweep = {0: {1: 0.87, 2: 1.77, 3: 27.48, 4: 38.13, 5: 38.95, 6: 23.86}, 1: {1: 1.06, 2: 3.21, 3: 86.34, 4: 72.63, 5: 197.85, 6: 468.28},
         2: {1: 0.94, 2: 1.82, 3: 52.75, 4: 185.1, 5: 815.59, 6: 1124.73}}
print("(1) SWEEP medians, D2-D6 at nominal centres, 10% errors")
for e, dd in sweep.items():
    lam = [DET_UM[d] for d in (2, 3, 4, 5, 6)]; f = [dd[d] for d in (2, 3, 4, 5, 6)]; er = [0.1 * x for x in f]
    for ak in (0.0, 2.0, 4.0):
        p, pe, rc = fit(lam, f, er, ak=ak)
        print(f"  ep{e} A_K={ak:.0f}: T={p[1]:7.0f} +-{pe[1]:5.0f} K  chi2r={rc:6.1f}  L(8kpc)={lum(p, ak, 8)[0]:9.0f} Lsun")
    out.setdefault('sweep', {})[e] = None
# (2) this check's points
P = json.load(open('out/spectra_points.json'))
def keep(q):
    if q['saturated_in_aperture'] or q['flux_cal'] is None: return False
    w = q['wave_um']
    if abs(w - 1.889) < 0.01 or abs(w - 4.039) < 0.01: return False
    if 4.24 < w < 4.30: return False
    return q['det'] >= 2
print("\n(2) THIS CHECK: deblended, calibrated per-exposure fluxes; errors max(stat, 10%)")
res2 = {}
for e in (1, 2):
    q = [x for x in P if x['ep'] == e and keep(x)]
    lam = np.array([x['wave_um'] for x in q]); f = np.array([x['flux_cal'] for x in q])
    er = np.maximum(np.array([x['err_mjy'] or 1 for x in q]), 0.10 * np.abs(f)); er = np.maximum(er, 0.7)
    for label, sel in (("1.3-5.0 um", lam > 1.3), ("2.2-5.0 um", lam > 2.2), ("3.8-5.0 um", lam > 3.8)):
        for ak in (0.0, 2.0, 3.0, 4.0):
            p, pe, rc = fit(lam[sel], f[sel], er[sel], ak=ak)
            L3, fb = lum(p, ak, 3); L8 = lum(p, ak, 8)[0]; L12 = lum(p, ak, 12)[0]
            res2[(e, label, ak)] = dict(T=p[1], Terr=pe[1], chi2r=rc, L3=L3, L8=L8, L12=L12, fbol=fb)
            print(f"  ep{e} {label} A_K={ak:.0f}: T={p[1]:6.0f} +-{pe[1]:4.0f} K chi2r={rc:6.2f}  Fbol={fb*1e3:.2e} erg/s/cm2  L: 3kpc {L3:8.0f}  8kpc {L8:8.0f}  12kpc {L12:8.0f} Lsun")
        p, pe, rc = fit(lam[sel], f[sel], er[sel], free_ak=True)
        print(f"  ep{e} {label} A_K free: T={p[1]:6.0f} +-{pe[1]:4.0f} K, A_K={p[2]:.2f} +-{pe[2]:.2f}, chi2r={rc:.2f}, L8={lum(p, p[2], 8)[0]:.0f} Lsun")
json.dump({f"{k[0]}|{k[1]}|{k[2]}": v for k, v in res2.items()}, open('out/sedfit.json', 'w'), indent=1)
