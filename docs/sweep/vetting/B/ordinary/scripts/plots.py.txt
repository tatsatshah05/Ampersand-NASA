"""Two PNGs with PIL: spectra per epoch (deblended, calibrated) and per-exposure positions."""
import json, math, numpy as np
from PIL import Image, ImageDraw
P = json.load(open('out/spectra_points.json'))
A = json.load(open('out/astro_points.json'))['points']
W, H = 1000, 620; L, R, T, B = 80, 30, 40, 70
im = Image.new('RGB', (W, H), 'white'); d = ImageDraw.Draw(im)
x0, x1 = 1.1, 5.05; y0, y1 = math.log10(0.3), math.log10(3000)
X = lambda w: L + (w - x0) / (x1 - x0) * (W - L - R)
Y = lambda f: H - B - (math.log10(max(f, 0.3)) - y0) / (y1 - y0) * (H - T - B)
for v in (1, 10, 100, 1000):
    d.line([(L, Y(v)), (W - R, Y(v))], fill=(225, 225, 225)); d.text((10, Y(v) - 6), f"{v} mJy", fill='black')
for w in (1.5, 2, 2.5, 3, 3.5, 4, 4.5, 5):
    d.line([(X(w), T), (X(w), H - B)], fill=(235, 235, 235)); d.text((X(w) - 8, H - B + 8), f"{w}", fill='black')
d.text((W // 2 - 60, H - 30), "wavelength (micrometres)", fill='black')
d.text((L, 10), "Candidate B, PSF flux at RA 253.18036 Dec -44.67701 (neighbour 10\" fitted out). grey: Aug 2025 3-sigma limits; blue: Mar 2026; red: Aug 2026; x = saturated", fill='black')
col = {1: (40, 90, 220), 2: (210, 40, 40)}
for p in P:
    if p['ep'] == 0 or p['det'] < 2 or p['flux_cal'] is None and not p['saturated_in_aperture']: continue
    f = p['flux_cal'] if p['flux_cal'] is not None else p['flux_mjy']; xx, yy = X(p['wave_um']), Y(f)
    if p['saturated_in_aperture']:
        d.line([(xx - 4, yy - 4), (xx + 4, yy + 4)], fill=col[p['ep']]); d.line([(xx - 4, yy + 4), (xx + 4, yy - 4)], fill=col[p['ep']])
    else:
        d.ellipse([xx - 3, yy - 3, xx + 3, yy + 3], fill=col[p['ep']])
for det, (lo, hi), lim in ((2, (1.16, 1.64), 0.7), (3, (1.62, 1.78), 1.9), (4, (2.43, 3.74), 3.6), (5, (3.83, 4.41), 3.8), (6, (4.41, 4.94), 4.0)):
    d.line([(X(lo), Y(lim)), (X(hi), Y(lim))], fill=(130, 130, 130), width=3)
for lab, w in (("Pa-a", 1.875), ("Br-a", 4.052), ("CO2 ice", 4.27), ("H2O ice", 3.05)):
    d.text((X(w) - 12, T + 8), lab, fill=(90, 90, 90)); d.line([(X(w), T + 20), (X(w), T + 30)], fill=(90, 90, 90))
im.save('out/B_spectra_by_epoch.png')
# positions
im = Image.new('RGB', (700, 700), 'white'); d = ImageDraw.Draw(im); C = 350; S = 60  # px per arcsec
for v in range(-5, 6):
    d.line([(C + v * S, 40), (C + v * S, 660)], fill=(235, 235, 235)); d.line([(40, C - v * S), (660, C - v * S)], fill=(235, 235, 235))
d.text((10, 10), "Offsets (arcsec) from the sweep position, corrected with 2MASS reference stars. East left. blue Mar 2026, red Aug 2026 (D4-D6)", fill='black')
def ep(m): return 0 if m < 61000 else (1 if m < 61200 else 2)
for p in A:
    if 'e' not in p or p['det'] < 4 or ep(p['mjd']) == 0 or p['n_ref'] < 3: continue
    xx, yy = C - p['e'] * S, C - p['n'] * S
    c = (40, 90, 220) if ep(p['mjd']) == 1 else (210, 40, 40)
    d.ellipse([xx - 2, yy - 2, xx + 2, yy + 2], fill=c)
for lab, e, n in (("2MASS J16524373 / Gaia G19.4 (+4.6,+0.1)", 4.6, 0.1), ("CatWISE (+3.5,-0.1)", 3.5, -0.1), ("GLIMPSE G341.0893 (+4.3,-0.4)", 4.3, -0.4), ("VVV Ks17.3 (-1.0,-2.2) rel. adopted", -1.21, -2.71)):
    xx, yy = C - e * S, C - n * S
    d.rectangle([xx - 4, yy - 4, xx + 4, yy + 4], outline='black'); d.text((xx - 60, yy + 8), lab, fill='black')
im.save('out/B_positions.png')
print('ok')
