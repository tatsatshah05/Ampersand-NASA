"""Small VizieR cone fetcher (TSV). Usage: viz.py <source> <ra> <dec> <radius_arcsec> <outfile> [extra -out cols]"""
import sys, requests, time
UA = {"User-Agent": "generic-astro-client/1.0"}
MIRRORS = ["https://vizier.cfa.harvard.edu/viz-bin/asu-tsv", "https://vizier.cds.unistra.fr/viz-bin/asu-tsv"]
def fetch(source, ra, dec, rad, out, extra=None):
    params = {"-source": source, "-c": f"{ra} {dec}", "-c.rs": str(rad), "-out.max": "5000", "-out.add": "_r,_p", "-sort": "_r"}
    if extra:
        params["-out"] = extra
    last = None
    for m in MIRRORS:
        for t in range(3):
            try:
                r = requests.get(m, params=params, headers=UA, timeout=90)
                if r.status_code == 200:
                    open(out, "w").write(f"# mirror {m}\n# params {params}\n" + r.text)
                    return r.text
                last = f"HTTP {r.status_code}"
            except Exception as e:
                last = repr(e)
            time.sleep(2)
    raise SystemExit(f"failed: {last}")
if __name__ == "__main__":
    a = sys.argv
    fetch(a[1], a[2], a[3], a[4], a[5], a[6] if len(a) > 6 else None)
