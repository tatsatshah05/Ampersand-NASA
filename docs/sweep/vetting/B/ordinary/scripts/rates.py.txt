import json, math, numpy as np
from astropy.time import Time
P = json.load(open('out/spectra_points.json'))
A = json.load(open('out/astro_points.json'))['points']
def ep(m): return 0 if m < 61000 else (1 if m < 61200 else 2)
print("Aug-Sep 2025 (epoch 0) forced fluxes at the adopted position:")
for d in range(1, 7):
    q = [p for p in P if p['ep'] == 0 and p['det'] == d]
    if not q: continue
    f = np.array([p['flux_mjy'] for p in q]); e = np.array([p['err_mjy'] for p in q])
    w = 1 / e**2; mean = np.sum(w * f) / np.sum(w); emean = 1 / math.sqrt(np.sum(w))
    print(f"  D{d}: n={len(q)}, weighted mean {mean:+.2f} +- {emean:.2f} mJy (scatter-based +-{np.std(f)/math.sqrt(len(f)):.2f}); 3-sigma upper limit {mean + 3*max(emean, np.std(f)/math.sqrt(len(f))):.1f} mJy; dates {Time(min(p['mjd'] for p in q),format='mjd').iso[:10]}..{Time(max(p['mjd'] for p in q),format='mjd').iso[:10]}")
def mag_rate(f1, f2, t1, t2):
    return -2.5 * math.log10(f2 / f1) / ((t2 - t1) / 30.44)
def T(s): return Time(s).mjd
print("\nRates (negative = brightening), magnitudes per month:")
rows = [
 ("4.9 um, 8 Mar -> 15/16 Mar 2026 (D6 4.93/4.97 vs 4.80/4.83/4.90)", np.mean([277.2, 298.0]), np.mean([451.4, 426.1, 428.7]), T('2026-03-08'), T('2026-03-15T12:00')),
 ("4.4-5.0 um, mid-Mar -> mid-Aug 2026 (D6 medians 14-16 Mar vs 8-14 Aug)", 410.0, 1015.0, T('2026-03-14T12:00'), T('2026-08-11')),
 ("3.9-4.2 um, 6-7 Mar -> 11-16 Aug (D5 3.87-4.22 medians)", None, None, None, None),
 ("~3.0 um, 28 Feb -> 25 Aug (D4 3.06 um: 26.5 -> 154.1)", 26.5, 154.1, T('2026-02-28'), T('2026-08-25')),
 ("2.27-2.40 um, 16 Mar -> 14 Aug (D3 2.268: 64.9 -> 2.398: 57.2)", 64.9, 57.2, T('2026-03-16'), T('2026-08-14')),
 ("1.6 um, 4 Mar -> 16-23 Aug (D2 1.617: 3.7 -> 1.62-1.63: <~1.5 at 3 sigma)", 3.7, 1.5, T('2026-03-04'), T('2026-08-20')),
 ("4.4-5.0 um, Aug-Sep 2025 upper limit (<5 mJy, 3 sigma) -> 8 Mar 2026 (~285)", 5.0, 285.0, T('2025-09-01'), T('2026-03-08')),
]
d5a = np.median([p['flux_cal'] for p in P if p['ep'] == 1 and p['det'] == 5 and 3.86 < p['wave_um'] < 4.23 and abs(p['wave_um'] - 4.039) > 0.01 and Time(p['mjd'], format='mjd').iso[5:10] in ('03-06', '03-07')])
d5b = np.median([p['flux_cal'] for p in P if p['ep'] == 2 and p['det'] == 5 and 3.86 < p['wave_um'] < 4.23])
rows[2] = ("3.9-4.2 um, 6-7 Mar -> 11-18 Aug (D5 medians %.0f -> %.0f)" % (d5a, d5b), d5a, d5b, T('2026-03-06T12:00'), T('2026-08-14'))
for lab, f1, f2, t1, t2 in rows:
    print(f"  {lab}: {mag_rate(f1, f2, t1, t2):+.2f} mag/month over {(t2-t1):.0f} d (factor {f2/f1:.2f})")
