"""Tolerant reader of a (possibly truncated) VizieR all-catalogue VOTable: per table, the name, description,
number of rows and the _r values (first column), with the _r unit from the FIELD (VizieR's default is arcmin)."""
import sys, re, html

t = open(sys.argv[1], encoding="utf-8", errors="replace").read()
rmax_arcsec = float(sys.argv[2]) if len(sys.argv) > 2 else 99
parts = re.split(r'(?=<TABLE ID=")', t)
out = []
complete = t.rstrip().endswith("</VOTABLE>")
for p in parts[1:]:
    m = re.match(r'<TABLE ID="[^"]*" name="([^"]*)"', p)
    name = m.group(1) if m else "?"
    d = re.search(r"<DESCRIPTION>(.*?)</DESCRIPTION>", p, re.S)
    desc = html.unescape(d.group(1).strip()) if d else ""
    f = re.search(r'<FIELD name="_r"[^>]*>', p)
    unit = re.search(r'unit="([^"]*)"', f.group(0)).group(1) if f and 'unit="' in f.group(0) else "arcsec(as -c.rs)"
    rows = re.findall(r"<TR>(.*?)</TR>", p, re.S)
    rs = []
    for r in rows:
        tds = re.findall(r"<TD>(.*?)</TD>", r, re.S)
        try:
            v = float(tds[0])
            v = v * 60 if unit.startswith("arcmin") else (v * 3600 if unit == "deg" else v)
            rs.append((v, [html.unescape(x)[:24] for x in tds[1:9]]))
        except (ValueError, IndexError):
            pass
    truncated = "</TABLE>" not in p
    out.append((min([x[0] for x in rs]) if rs else 1e9, name, desc, len(rows), rs, truncated))
print(f"file complete: {complete}; tables: {len(out)}; tables with rows: {sum(1 for x in out if x[3])}")
print("last table in file:", out[-1][1] if out else None)
for rmin, name, desc, n, rs, tr in sorted(out):
    if n and rmin <= rmax_arcsec:
        best = sorted(rs)[0] if rs else (None, [])
        print(f"{rmin:6.2f}\" {name:28s} n={n:3d} {'(truncated)' if tr else ''} {desc[:60]} | {best[1]}")
