"""Is the Aug-Sep 2025 light at the target just the wing of the steady bright neighbour 17 arcsec away?
For each D5/D6 cutout: the zodi-subtracted pixel value at the target (bilinear), and the neighbour's profile at the
same distance (17") sampled at 24 azimuths, excluding +-60 deg around the target direction. Also a local sky
(median of a ring 8-12 px around the target)."""
import sys, glob, os, json, warnings
import numpy as np
warnings.filterwarnings("ignore")
from astropy.io import fits
from astropy.wcs import WCS
from astropy.coordinates import SkyCoord
from scipy.ndimage import map_coordinates

CUT = "<repository>/backend/cache/sweep/cutouts/250d2b7784"
T = SkyCoord(270.80362, -23.46745, unit="deg")
NB = SkyCoord(270.80192, -23.46297, unit="deg")
out = {}
for f in sorted(glob.glob(os.path.join(CUT, "*.fits"))):
    det = int(os.path.basename(f).split("_D")[-1][0])
    if det not in (5, 6):
        continue
    with fits.open(f) as h:
        hd = h["IMAGE"].header
        img = h["IMAGE"].data.astype(float) - h["ZODI"].data.astype(float)
        flg = h["FLAGS"].data
    w = WCS(hd)
    mjd = hd["MJD-AVG"]
    ep = "2025" if mjd < 60950 else ("2026a" if mjd < 61150 else "2026b")
    xt, yt = (float(v) for v in w.world_to_pixel(T))
    xn, yn = (float(v) for v in w.world_to_pixel(NB))
    if not (12 <= xt < img.shape[1] - 12 and 12 <= yt < img.shape[0] - 12):
        continue
    yy, xx = np.mgrid[0:img.shape[0], 0:img.shape[1]]
    rr = np.hypot(xx - xt, yy - yt)
    sky = np.nanmedian(img[(rr > 8) & (rr < 12)])
    vt = float(map_coordinates(np.nan_to_num(img), [[yt], [xt]], order=1)[0]) - sky
    dx, dy = xt - xn, yt - yn
    r = np.hypot(dx, dy); a0 = np.arctan2(dy, dx)
    vals = []
    for a in np.linspace(0, 2 * np.pi, 24, endpoint=False):
        if abs(np.angle(np.exp(1j * (a - a0)))) < np.radians(60):
            continue
        x, y = xn + r * np.cos(a), yn + r * np.sin(a)
        vals.append(float(map_coordinates(np.nan_to_num(img), [[y], [x]], order=1)[0]) - sky)
    vn = float(map_coordinates(np.nan_to_num(img), [[yn], [xn]], order=1)[0]) - sky
    out.setdefault((ep, det), []).append((vt, float(np.median(vals)), float(np.percentile(vals, 16)), float(np.percentile(vals, 84)), vn, r))
print("epoch det  n   target-sky   neighbour-wing at same radius (median [16-84%])   neighbour peak-sky   radius px")
for k in sorted(out):
    a = np.array(out[k])
    print(f"{k[0]:6s} D{k[1]} {len(a):3d}  {np.median(a[:,0]):9.1f}   {np.median(a[:,1]):9.1f} [{np.median(a[:,2]):7.1f} {np.median(a[:,3]):7.1f}]   {np.median(a[:,4]):9.1f}   {np.median(a[:,5]):.2f}")
