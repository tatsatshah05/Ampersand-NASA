"""Position checks against downloaded transient lists: ASAS-SN transients page, CBAT TOCP (TCP J-names),
and any page whose rows carry J-names or decimal coordinates. Target plus controls."""
import sys, re, html
import numpy as np

T = {"target": (270.80362, -23.46745), "V7994 Sgr": (270.96983, -31.45739), "V462 Lup": (227.01362, -40.14308),
     "V0691 Ser": (279.74438, -3.86389)}


def sep(a, b):
    r0, d0, r1, d1 = map(np.radians, (a[0], a[1], b[0], b[1]))
    c = np.sin(d0) * np.sin(d1) + np.cos(d0) * np.cos(d1) * np.cos(r1 - r0)
    return float(np.degrees(np.arccos(min(1, max(-1, c)))) * 3600)


def strip(s):
    s = re.sub(r"<!--.*?-->", " ", s, flags=re.S)
    return " ".join(html.unescape(re.sub(r"<[^>]+>", " ", s)).split())


def report(entries, label):
    print(f"== {label}: {len(entries)} entries with positions")
    for n, p in T.items():
        d = sorted((sep(p, (e[0], e[1])), e[2]) for e in entries)
        within = [x for x in d if x[0] < 3600]
        print(f"  {n}: within 60 arcsec: {sum(1 for x in d if x[0] < 60)}; within 1 deg: {len(within)}")
        for x in d[:3]:
            print(f"     {x[0]:9.1f} arcsec  {x[1][:230]}")


def asassn(path):
    t = open(path, encoding="utf-8", errors="replace").read()
    out = []
    for row in t.split("<tr")[1:]:
        m = re.search(r"sdss3\.org/dr9/en/tools/chart/navi\.asp\?ra=([-\d.]+)&dec=([-\d.]+)", row)
        if not m:
            m = re.search(r"-c=([-\d.]+)\+([-\d.]+)", row)
        if m:
            out.append((float(m.group(1)), float(m.group(2)), strip(row)))
    report(out, "ASAS-SN transients page")


def tcp_names(path, label):
    t = open(path, encoding="utf-8", errors="replace").read()
    out = []
    seen = set()
    for m in re.finditer(r"J(\d{2})(\d{2})(\d{2})(\d{2})([+-])(\d{2})(\d{2})(\d{2,3})", t):
        hh, mm, ss, cs, sg, dd, dm, ds = m.groups()
        ra = 15 * (int(hh) + int(mm) / 60 + (int(ss) + int(cs) / 100) / 3600)
        dsec = int(ds) / (10 if len(ds) == 3 else 1)
        dec = (1 if sg == "+" else -1) * (int(dd) + int(dm) / 60 + dsec / 3600)
        if m.group(0) in seen:
            continue
        seen.add(m.group(0))
        ctx = strip(t[max(0, m.start() - 200): m.end() + 300])
        out.append((ra, dec, m.group(0) + " | " + ctx))
    report(out, label + " (J-names decoded)")


if __name__ == "__main__":
    kind, path = sys.argv[1], sys.argv[2]
    if kind == "asassn":
        asassn(path)
    else:
        tcp_names(path, kind)
