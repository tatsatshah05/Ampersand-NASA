"""Picture: per epoch, the median of all D6 cutouts resampled onto a common sky grid around the target, with the
target (circle, radius 2.5 px = the sweep's aperture) and the steady catalogued neighbour 17" away (square) marked."""
import glob, os, warnings
import numpy as np
warnings.filterwarnings("ignore")
from astropy.io import fits
from astropy.wcs import WCS
from astropy.coordinates import SkyCoord
from scipy.ndimage import map_coordinates
from PIL import Image, ImageDraw

CUT = "<repository>/backend/cache/sweep/cutouts/250d2b7784"
OUT = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "candidateA_D6_epochs_neighbour.png")
T = SkyCoord(270.80362, -23.46745, unit="deg")
NB = SkyCoord(270.80192, -23.46297, unit="deg")
half, step = 60.0, 2.0   # arcsec
offs = np.arange(-half, half + step, step)
E, N = np.meshgrid(offs, offs)  # east (+), north (+)
dec = T.dec.deg + N / 3600
ra = T.ra.deg + E / 3600 / np.cos(np.radians(T.dec.deg))
stacks = {}
for f in sorted(glob.glob(os.path.join(CUT, "*_D6.fits"))):
    with fits.open(f) as h:
        hd = h["IMAGE"].header
        img = h["IMAGE"].data.astype(float) - h["ZODI"].data.astype(float)
    w = WCS(hd)
    mjd = hd["MJD-AVG"]
    ep = "Aug-Sep 2025" if mjd < 60950 else ("Mar-Apr 2026" if mjd < 61150 else "Aug-Sep 2026")
    x, y = w.world_to_pixel(SkyCoord(ra.ravel(), dec.ravel(), unit="deg"))
    if np.nanmin(x) < 0 or np.nanmin(y) < 0 or np.nanmax(x) > img.shape[1] - 1 or np.nanmax(y) > img.shape[0] - 1:
        continue
    v = map_coordinates(np.nan_to_num(img), [y, x], order=1).reshape(E.shape)
    v -= np.nanmedian(v)
    stacks.setdefault(ep, []).append(v)
nb_e = (NB.ra.deg - T.ra.deg) * np.cos(np.radians(T.dec.deg)) * 3600
nb_n = (NB.dec.deg - T.dec.deg) * 3600
S = 4  # screen pixels per grid cell
n = len(offs)
panels = []
def to_px(e, nn):  # east to the left, north up
    return ((half - e) / step * S, (half - nn) / step * S)
for ep in ["Aug-Sep 2025", "Mar-Apr 2026", "Aug-Sep 2026"]:
    m = np.median(stacks[ep], axis=0)
    v = np.clip(np.arcsinh(np.clip(m, 0, None) / 5) / np.arcsinh(600 / 5), 0, 1)
    a = (255 * v[::-1, ::-1]).astype(np.uint8)   # flip: north up, east left
    im = Image.fromarray(a).resize((n * S, n * S), Image.NEAREST).convert("RGB")
    d = ImageDraw.Draw(im)
    cx, cy = to_px(0, 0); r = 2.5 * 6.15 / step * S
    d.ellipse([cx - r, cy - r, cx + r, cy + r], outline=(230, 60, 60), width=2)
    nx, ny = to_px(nb_e, nb_n); q = 5 / step * S
    d.rectangle([nx - q, ny - q, nx + q, ny + q], outline=(60, 200, 230), width=2)
    d.text((4, 4), f"D6 {ep} (median of {len(stacks[ep])})", fill=(255, 255, 0))
    panels.append(im)
W = sum(p.width for p in panels) + 20; H = panels[0].height + 40
canvas = Image.new("RGB", (W, H), (20, 20, 20))
x = 0
for p in panels:
    canvas.paste(p, (x, 0)); x += p.width + 10
ImageDraw.Draw(canvas).text((4, H - 34), "Candidate A RA 270.80362 Dec -23.46745, 120 arcsec boxes, north up east left. Red circle: sweep aperture r=2.5 px at target.", fill=(220, 220, 220))
ImageDraw.Draw(canvas).text((4, H - 18), "Cyan square: AllWISE J180312.46-232746.7 (W2 6.49 mag, ~0.44 Jy, steady), 17 arcsec away.", fill=(220, 220, 220))
canvas.save(OUT)
print(OUT, {k: len(v) for k, v in stacks.items()})
