import json, numpy as np
from collections import defaultdict
R=json.load(open("fit2.json"))
def mad(a):
    a=np.asarray(a,float); return 1.4826*np.median(np.abs(a-np.median(a))) if a.size else np.nan
g=defaultdict(list)
for r in R:
    if r.get("status")=="done": g[(r["det"],r["epoch"])].append(r)
print("det ep  N | ap1.5(A') ok: n  med  mad | ap2.5(A') ok: n med mad | nb_in_ap2.5 | ap2.5-nb (all finite) med | gaussA' med mad  | gaussNb med | fwhm chi2")
for k in sorted(g):
    rs=g[k]
    a15=[r["ap1.5"] for r in rs if r["ap1.5_status"]=="ok"]
    a25=[r["ap2.5"] for r in rs if r["ap2.5_status"]=="ok"]
    a25c=[r["ap2.5"]-r.get("nb_in_ap2.5",0) for r in rs if r["ap2.5_status"]=="ok" and r.get("fit_ok")]
    nb=[r.get("nb_in_ap2.5",np.nan) for r in rs if r.get("fit_ok")]
    ga=[r["A_gauss"] for r in rs if r.get("fit_ok")]
    gae=[r["A_gauss_err"] for r in rs if r.get("fit_ok")]
    gn=[r["N_gauss"] for r in rs if r.get("fit_ok")]
    fw=[r["fwhm_arcsec"] for r in rs if r.get("fit_ok")]
    c2=[r["chi2_dof"] for r in rs if r.get("fit_ok")]
    print(f"{k[0]} {k[1]} {len(rs):3d} | {len(a15):2d} {np.median(a15) if a15 else np.nan:8.1f} {mad(a15):6.1f} | {len(a25):2d} {np.median(a25) if a25 else np.nan:8.1f} {mad(a25):6.1f} | {np.nanmedian(nb):7.1f} | {np.median(a25c) if a25c else np.nan:8.1f} | {np.median(ga):8.1f} {mad(ga):6.1f} (err {np.median(gae):5.1f}) | {np.median(gn):8.1f} | {np.median(fw):5.1f} {np.median(c2):6.1f}")
