import json, sys
from collections import defaultdict
import numpy as np
from pathlib import Path
D = Path(__file__).resolve().parent
R = json.load(open(D / "measurements.json"))
EP = {1: "Aug-Sep 2025", 2: "Mar-Apr 2026", 3: "Aug-Sep 2026"}
# bright neighbour (AllWISE J180312.46-232746.7 / 2MASS 18031245-2327467) offset from nominal
import math
RA0, DEC0 = 270.80362, -23.46745
SRA, SDEC = 270.801899, -23.462976
se = (SRA - RA0) * math.cos(math.radians(DEC0)) * 3600; sn = (SDEC - DEC0) * 3600
print(f"bright 2MASS/AllWISE star at east {se:.2f} north {sn:.2f} arcsec from nominal")

def mad(a):
    a = np.asarray(a); return 1.4826 * np.median(np.abs(a - np.median(a))) if a.size else np.nan

g = defaultdict(list)
for r in R: g[(r["det"], r["epoch"])].append(r)
print(f"{'det':>3} {'ep':>2} {'N':>3} {'ok':>3} {'sat':>3} {'flg':>3} {'edge':>4} | {'med':>8} {'mad':>7} {'min':>8} {'max':>8} | {'empty_sig':>9} | {'cenE':>6} {'cenN':>6} {'scat':>5} | {'fwhm':>5} {'cmpF':>5} {'ar':>4} | wave range")
for k in sorted(g):
    rs = g[k]
    st = defaultdict(int)
    for r in rs: st[r["status"]] += 1
    ok = [r for r in rs if r["status"] == "ok"]
    fl = np.array([r["flux"] for r in ok])
    es = np.array([r.get("empty_sig", np.nan) for r in ok])
    fits_ = [r["fit"] for r in rs if r.get("fit") and r["status"] in ("ok", "saturated", "flagged")]
    cen = np.array([[f["east"], f["north"]] for f in fits_]) if fits_ else np.zeros((0, 2))
    fw = np.array([f["fwhm_arcsec"] for f in fits_]) if fits_ else np.array([])
    ar = np.array([f["axis_ratio"] for f in fits_]) if fits_ else np.array([])
    comp = []
    for r in rs:
        for c in r.get("comps", []):
            comp.append(c["fwhm_px"] * 6.15)
    waves = [r.get("wave") for r in rs if r.get("wave")]
    print(f"{k[0]:>3} {k[1]:>2} {len(rs):>3} {st['ok']:>3} {st['saturated']:>3} {st['flagged']:>3} {st['edge']+st['no_sky']:>4} | "
          f"{(np.median(fl) if fl.size else np.nan):8.1f} {mad(fl):7.1f} {(fl.min() if fl.size else np.nan):8.1f} {(fl.max() if fl.size else np.nan):8.1f} | "
          f"{np.nanmedian(es) if es.size else np.nan:9.2f} | "
          f"{(np.median(cen[:,0]) if len(cen) else np.nan):6.2f} {(np.median(cen[:,1]) if len(cen) else np.nan):6.2f} "
          f"{(np.sqrt(mad(cen[:,0])**2+mad(cen[:,1])**2) if len(cen)>2 else np.nan):5.2f} | "
          f"{(np.median(fw) if fw.size else np.nan):5.1f} {(np.median(comp) if comp else np.nan):5.1f} {(np.median(ar) if ar.size else np.nan):4.2f} | "
          f"{min(waves) if waves else 0:.3f}-{max(waves) if waves else 0:.3f}")
if "-v" in sys.argv:
    for r in sorted(R, key=lambda r: (r["det"], r["mjd"] if "mjd" in r else 0)):
        f = r.get("fit") or {}
        print(f"D{r['det']} e{r['epoch']} {r['obs_id']:<20} mjd {r.get('mjd',0):.3f} w {r.get('wave',0):.3f} st {r['status']:<9} "
              f"F {r.get('flux', float('nan')):8.1f} Fany {r.get('flux_any', float('nan')):8.1f} pk {r.get('peak_mjysr', float('nan')):8.1f} "
              f"det({r.get('detx',0):6.0f},{r.get('dety',0):6.0f}) pa {r.get('pa_x',0):5.1f} "
              f"cen({f.get('east', float('nan')):6.2f},{f.get('north', float('nan')):6.2f}) fw {f.get('fwhm_arcsec', float('nan')):5.1f} "
              f"es {r.get('empty_sig', float('nan')):6.2f} core {','.join(r.get('flags_core', []))} | ap {','.join(r.get('flags_ap', []))}")
