import json, numpy as np, math
R=json.load(open("measurements.json"))
RA0,DEC0=270.80362,-23.46745
def sel(det, ep, near):
    out=[]
    for r in R:
        f=r.get("fit")
        if r["det"] in det and r["epoch"] in ep and f and r["status"] in ("ok","flagged","saturated"):
            if math.hypot(f["east"]-near[0], f["north"]-near[1])<2.0: out.append(r)
    return out
for name,near in [("A' (new source)",(1.5,-2.65)),("neighbour star",(-5.6,15.9))]:
    for det,ep in [((6,),(2,)),((6,),(3,)),((5,),(2,)),((5,),(3,)),((3,),(2,)),((3,),(3,)),((2,),(2,)),((4,),(2,3)),((6,),(1,)),((5,),(1,)),((2,),(1,)),((2,),(3,)),((3,4),(1,3)),((1,),(1,2,3))]:
        s=sel(det,ep,near)
        if not s: continue
        ra=np.array([r["fit"]["ra"] for r in s]); de=np.array([r["fit"]["dec"] for r in s])
        e=np.array([r["fit"]["east"] for r in s]); n=np.array([r["fit"]["north"] for r in s])
        ee=np.array([r["fit"]["err_arcsec"] for r in s])
        t=np.array([r["mjd"] for r in s])
        slope = (np.polyfit(t-t.mean(), e,1)[0], np.polyfit(t-t.mean(), n,1)[0]) if np.ptp(t)>0.5 and len(s)>3 else (np.nan,np.nan)
        print(f"{name:16s} D{det} ep{ep}: n={len(s):2d} RA {np.median(ra):.6f} Dec {np.median(de):.6f}  E {np.median(e):6.2f}+-{np.std(e):.2f} N {np.median(n):6.2f}+-{np.std(n):.2f}  fiterr {np.median(ee):.2f}  drift E {slope[0]:+.3f} N {slope[1]:+.3f} \"/day over {np.ptp(t):.1f} d")
