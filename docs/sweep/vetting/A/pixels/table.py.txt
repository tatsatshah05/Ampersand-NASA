"""Per detector and visit: (a) plain r=2.5 px aperture at the sweep's position (the sweep's method, own code),
(b) this check's crowded-field flux at A' (neighbour modelled), clean exposures only. Writes results_table.tsv."""
import json, numpy as np
from collections import defaultdict
M = json.load(open("measurements.json"))
S = [l.rstrip("\n").split("\t") for l in open("spectrum_table.tsv")][1:]
def mad(v): v = np.asarray(v, float); return 1.4826 * np.median(np.abs(v - np.median(v))) if v.size else float("nan")
a = defaultdict(list); n_all = defaultdict(int)
for r in M:
    n_all[(r["det"], r["epoch"])] += 1
    if r["status"] == "ok": a[(r["det"], r["epoch"])].append(r["flux"])
b = defaultdict(list); bs = defaultdict(int)
for p in S:
    ep, det, flux, sat = int(p[0]), int(p[1]), float(p[5]), p[7] == "1"
    if sat: bs[(det, ep)] += 1
    else: b[(det, ep)].append(flux)
with open("results_table.tsv", "w") as f:
    f.write("detector\tvisit\texposures\tsweep_pos_ap2.5_n_clean\tsweep_pos_median_mJy\tsweep_pos_MAD\tA'_n_clean\tA'_n_saturated_at_A'\tA'_median_mJy\tA'_MAD\n")
    for k in sorted(n_all):
        x, y = a[k], b[k]
        f.write(f"{k[0]}\t{k[1]}\t{n_all[k]}\t{len(x)}\t{np.median(x) if x else float('nan'):.1f}\t{mad(x):.1f}\t{len(y)}\t{bs[k]}\t{np.median(y) if y else float('nan'):.1f}\t{mad(y):.1f}\n")
print(open("results_table.tsv").read())
