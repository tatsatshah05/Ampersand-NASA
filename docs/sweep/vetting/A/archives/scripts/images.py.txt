photometry at P calibrated against catalogued stars in the same image.

Outputs: png/prior_images.png (panels), tables/forced_phot.json, printed table."""

# common grid: 1 arcsec/px, 120 x 120 arcsec, north up, east left
N, SC = 121, 1.0
gw = WCS(naxis=2)
gw.wcs.ctype = ["RA---TAN", "DEC--TAN"]
gw.wcs.crval = [P.ra.deg, P.dec.deg]
gw.wcs.crpix = [(N + 1) / 2, (N + 1) / 2]
gw.wcs.cdelt = [-SC / 3600, SC / 3600]


def load(path, ext=0):
    with fits.open(path) as h:
        hd = h[ext].header
        d = h[ext].data.astype(float)
    if d.ndim > 2:
        d = d[0]
    return d, hd, WCS(hd).celestial


def aperture(data, wcs, coord, r_as, rin_as, rout_as):
    """Sum in a circle minus median sky in an annulus (native pixels), returns (flux_sum, sky_per_px, npix, pixscale)."""
    ps = np.sqrt(abs(np.linalg.det(wcs.pixel_scale_matrix))) * 3600
    if wcs.wcs.ctype[0].startswith("GLON"):
        g = coord.galactic
        x, y = wcs.all_world2pix(g.l.deg, g.b.deg, 0)
    else:
        x, y = wcs.all_world2pix(coord.ra.deg, coord.dec.deg, 0)
    yy, xx = np.mgrid[0:data.shape[0], 0:data.shape[1]]
    rr = np.hypot(xx - x, yy - y) * ps
    ann = (rr >= rin_as) & (rr < rout_as) & np.isfinite(data)
    if ann.sum() < 10:
        return np.nan, np.nan, 0, ps
    _, sky, _ = sigma_clipped_stats(data[ann], sigma=3, maxiters=5)
    ap = (rr < r_as) & np.isfinite(data)
    if ap.sum() == 0 or (rr < r_as).sum() != ap.sum():
        return np.nan, sky, int(ap.sum()), ps
    return float(np.sum(data[ap] - sky)), float(sky), int(ap.sum()), ps


def empty_apertures(data, wcs, centre, r_as, rin_as, rout_as, ring=(15, 60), n=60, avoid=()):
    """Same aperture at n positions on a ring of radii ring around centre (avoiding listed coords by 2r)."""
    rng = np.random.default_rng(1)
    vals = []
    tries = 0
    while len(vals) < n and tries < 2000:
        tries += 1
        rad = rng.uniform(*ring)
        ang = rng.uniform(0, 2 * np.pi)
        c = centre.directional_offset_by(ang * 180 / np.pi * __import__("astropy.units", fromlist=["deg"]).deg,
                                         rad * __import__("astropy.units", fromlist=["arcsec"]).arcsec)
        if any(c.separation(a).arcsec < 2 * r_as + 2 for a in avoid):
            continue
        f, _, npx, _ = aperture(data, wcs, c, r_as, rin_as, rout_as)
        if np.isfinite(f):
            vals.append(f)
    vals = np.array(vals)
    med = np.median(vals)
    mad = 1.4826 * np.median(np.abs(vals - med))
    return med, mad, len(vals)


def calib(data, wcs, stars, r_as, rin_as, rout_as):
    """stars: list of (SkyCoord, flux_mJy). Returns median mJy-per-unit and the individual ratios."""
    rat = []
    for c, f in stars:
        a, _, _, _ = aperture(data, wcs, c, r_as, rin_as, rout_as)
        if np.isfinite(a) and a > 0 and np.isfinite(f) and f > 0:
            rat.append(f / a)
    rat = np.array(rat)
    return (np.median(rat) if len(rat) else np.nan), rat


def tbl(name):
    return ascii.read(open(os.path.join(TAB, name)).read(), format="ipac")


# ------------- catalogue stars for calibration
tm = tbl("gator_cand_fp_psc_r30.tbl")
glm = tbl("gator_cand_glimpse2_v2arc_r30.tbl")
aw = tbl("gator_cand_allwise_p3as_psd_r30.tbl")
ZP = {"J": 1594e3, "H": 1024e3, "K": 666.7e3, "W1": 309.540e3, "W2": 171.787e3, "W3": 31.674e3, "W4": 8.363e3,
      "I1": 280.9e3, "I2": 179.7e3, "I3": 115.0e3, "I4": 64.13e3}   # mJy for mag 0 (2MASS Cohen+03; WISE Jarrett+11; IRAC handbook)


def mag2mjy(m, band):
    return ZP[band] * 10 ** (-0.4 * m)


def stars_2mass(col, band):
    out = []
    for r in tm:
        m = r[col]
        if np.ma.is_masked(m):
            continue
        q = r["ph_qual"]["JHK".index(band)]
        if q != "A":
            continue
        out.append((SkyCoord(r["ra"], r["dec"], unit="deg"), mag2mjy(float(m), band)))
    return out


def stars_glm(col, band):
    out = []
    for r in glm:
        m = r[col]
        if np.ma.is_masked(m) or not np.isfinite(float(m)):
            continue
        out.append((SkyCoord(r["ra"], r["dec"], unit="deg"), mag2mjy(float(m), band)))
    return out


def stars_aw(col, band, snrcol):
    out = []
    for r in aw:
        m, s = r[col], r[snrcol]
        if np.ma.is_masked(m) or float(s) < 10:
            continue
        out.append((SkyCoord(r["ra"], r["dec"], unit="deg"), mag2mjy(float(m), band)))
    return out

results = []
panels = []
for label, path, ext, band, rap, rin, rout, stars, epoch in imgs:
    if not os.path.exists(path):
        print("missing", path)
        continue
    d, hd, w = load(path, ext)
    # crop big images (MIPSGAL mosaic) around P first
    if d.shape[0] > 1000:
        if w.wcs.ctype[0].startswith("GLON"):
            x, y = w.all_world2pix(P.galactic.l.deg, P.galactic.b.deg, 0)
        else:
            x, y = w.all_world2pix(P.ra.deg, P.dec.deg, 0)
        x, y = int(x), int(y)
        d = d[y - 150:y + 151, x - 150:x + 151]
        w = w[y - 150:y + 151, x - 150:x + 151]
    k, rats = calib(d, w, stars, rap, rin, rout)
    fP, sky, npx, ps = aperture(d, w, P, rap, rin, rout)
    fC, _, _, _ = aperture(d, w, C, rap, rin, rout)
    fW, _, _, _ = aperture(d, w, W, rap, rin, rout)
    med, mad, ne = empty_apertures(d, w, P, rap, rin, rout, ring=(rap * 2 + 3, 60), avoid=[W])
    rec = {"image": label, "epoch": epoch, "band": band, "aperture_arcsec": rap, "n_cal_stars": int(len(rats)),
           "cal_ratio_spread": [float(np.min(rats)), float(np.max(rats))] if len(rats) else None,
           "mJy_per_unit": float(k) if np.isfinite(k) else None,
           "P_mJy": float(fP * k) if np.isfinite(k) else None, "C_mJy": float(fC * k) if np.isfinite(k) else None,
           "W_mJy_measured": float(fW * k) if np.isfinite(k) else None,
           "empty_ap_median_mJy": float(med * k) if np.isfinite(k) else None, "empty_ap_sigma_mJy": float(mad * k) if np.isfinite(k) else None,
           "n_empty": ne}
    if rec["P_mJy"] is not None:
        rec["upper_limit_3sigma_mJy"] = max(rec["P_mJy"] - rec["empty_ap_median_mJy"], 0) + 3 * rec["empty_ap_sigma_mJy"]
    results.append(rec)
    # reproject onto common grid for the picture
    if w.wcs.ctype[0].startswith("GLON"):
        # build equatorial view through pixel->world->pixel: reproject handles celestial frames
        pass
    arr, _ = reproject_interp((d, w), gw, shape_out=(N, N))
    panels.append((f"{label} ({epoch})", arr))
