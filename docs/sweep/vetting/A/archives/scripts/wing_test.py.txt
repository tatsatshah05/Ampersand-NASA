"""Is the sweep's 'Aug 2025: 153 mJy at C' the wing of W?  Use the project's own aperture photometry
(backend.photometry.measure, r = 2.5 px) on the cached Aug-Sep 2025 D5/D6 cutouts at:
  C (sweep position, 17.0" from W), P (2026 centroid, 19.8" from W), and 8 control points 17.0" from W
  at other position angles (same distance from W, away from C)."""
import os, sys, glob, json
import numpy as np
sys.path.insert(0, "<repository>/.claude/worktrees/agent-afd7766a5a5ec349f")
os.environ.setdefault("SPHEREX_CACHE_DIR", "<repository>/backend/cache/sweep")
from backend import photometry
from astropy.coordinates import SkyCoord
import astropy.units as u
from astropy.io import fits

CUT = "<repository>/backend/cache/sweep/cutouts/250d2b7784"
W = SkyCoord(270.8020277, -23.4629541, unit="deg")
C = SkyCoord(270.80362, -23.46745, unit="deg")
P = SkyCoord(270.80409, -23.46819, unit="deg")
paC = W.position_angle(C).deg
pts = {"C": C, "P": P}
for pa in (paC + 45 * k for k in range(1, 8)):
    pts[f"W+17@{pa % 360:.0f}"] = W.directional_offset_by(pa * u.deg, W.separation(C))
out = {}
for det in (6, 5):
    for ep, lo, hi in (("2025-08/09", 0, 61000), ("2026-08/09", 61200, 99999)):
        vals = {k: [] for k in pts}
        for f in sorted(glob.glob(f"{CUT}/*_D{det}.fits")):
            mjd = fits.getheader(f, "IMAGE")["MJD-AVG"]
            if not (lo <= mjd < hi):
                continue
            for k, c in pts.items():
                m = photometry.measure(__import__("pathlib").Path(f), c.ra.deg, c.dec.deg)
                if m["status"] == "ok":
                    vals[k].append(m["flux_mjy"])
        out[f"D{det} {ep}"] = {k: (float(np.median(v)) if v else None, len(v)) for k, v in vals.items()}
        print(f"D{det} {ep}: " + "  ".join(f"{k}={np.median(v):.0f}(n{len(v)})" if v else f"{k}=--" for k, v in vals.items()))
json.dump(out, open("docs/sweep/vetting/A/archives/tables/wing_test.json", "w"), indent=1)
