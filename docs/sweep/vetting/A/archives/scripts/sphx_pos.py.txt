"""Where, exactly, is the SPHEREx source in each epoch, relative to the candidate position and to
the bright WISE source 17 arcsec NNW?  Reads the sweep follow-up's cached cutouts (read only).

For each cutout (detectors 2-6): image - zodi, flagged/saturated pixels masked, local sky from a
14-20 px ring (sigma clipped).  Then
  (1) a free fit: one circular Gaussian (FWHM = header PSF_FWHM) + constant, position free, in an
      11x11 px box centred on the candidate;
  (2) a fixed-position fit: Gaussians at the candidate (C) and at the WISE source (W) + constant,
      amplitudes free (linear least squares).
Fluxes are Gaussian-model fluxes in mJy (model integral), only for comparing C with W.
"""
import os, sys, glob, json
import numpy as np
from astropy.io import fits
from astropy.wcs import WCS
from astropy.wcs.utils import proj_plane_pixel_area
from astropy.stats import sigma_clipped_stats
from astropy.coordinates import SkyCoord
from scipy.optimize import least_squares

CUT = "<repository>/backend/cache/sweep/cutouts/250d2b7784"
OUT = "docs/sweep/vetting/A/archives/tables"
C = (270.80362, -23.46745)
W = (270.8020277, -23.4629541)   # CatWISE2020 J180312.48-232746.6 (W2 6.725)
W_allwise = (270.8019175, -23.4629914)

# flag bits as in the project's pipeline (read, not imported, to keep this standalone)
sys.path.insert(0, "<repository>/.claude/worktrees/agent-afd7766a5a5ec349f")
os.environ.setdefault("SPHEREX_CACHE_DIR", "<repository>/backend/cache/sweep")
from backend import pipeline  # noqa: E402
BAD = pipeline._mask("SUR_ERROR", "NONFUNC", "MISSING_DATA", "HOT", "COLD", "NONLINEAR", "PERSIST",
                     "BLOOM", "SNOWBALL", "HALO", "SATELLITE_HALO", "OVERFLOW")
SKYBAD = BAD | pipeline._mask("OUTLIER", "SOURCE", "TRANSIENT")


def epoch_of(mjd):
    if mjd < 61000:
        return "2025-08/09"
    if mjd < 61200:
        return "2026-03/04"
    return "2026-08/09"


def gauss(xx, yy, x0, y0, sig):
    return np.exp(-0.5 * ((xx - x0) ** 2 + (yy - y0) ** 2) / sig ** 2) / (2 * np.pi * sig ** 2)


rows = []
for path in sorted(glob.glob(os.path.join(CUT, "*.fits"))):
    det = int(path[-6])
    if det < 2:
        continue
    with fits.open(path) as h:
        hd = h["IMAGE"].header
        img = h["IMAGE"].data.astype(float) - h["ZODI"].data.astype(float)
        flg = h["FLAGS"].data
        wave = h["WAVE"].data
    wcs = WCS(hd)
    mjd = float(hd["MJD-AVG"])
    fwhm_as = float(hd.get("PSF_FWHM", 7.0))
    pixas = np.sqrt(proj_plane_pixel_area(wcs)) * 3600
    sig = fwhm_as / pixas / 2.3548
    to_mjy = proj_plane_pixel_area(wcs) * (np.pi / 180) ** 2 * 1e9
    cx, cy = [float(v) for v in wcs.all_world2pix(C[0], C[1], 0)]
    wx, wy = [float(v) for v in wcs.all_world2pix(W[0], W[1], 0)]
    ny, nx = img.shape
    if not (22 < cx < nx - 22 and 22 < cy < ny - 22):
        continue
    yy, xx = np.mgrid[0:ny, 0:nx]
    r = np.hypot(xx - cx, yy - cy)
    ring = (r >= 14) & (r < 20) & np.isfinite(img) & ((flg & SKYBAD) == 0)
    if ring.sum() < 60:
        ring = (r >= 14) & (r < 20) & np.isfinite(img) & ((flg & BAD) == 0)
    _, sky, ssig = sigma_clipped_stats(img[ring], sigma=3, maxiters=5)
    d = img - sky
    box = (np.abs(xx - cx) <= 5) & (np.abs(yy - cy) <= 5)
    good = box & np.isfinite(d) & ((flg & BAD) == 0)
    nbad = int((box & ~good).sum())
    # how many flagged pixels within 2 px of C and of W
    badC = int(((np.hypot(xx - cx, yy - cy) <= 2) & ~(np.isfinite(d) & ((flg & BAD) == 0))).sum())
    badW = int(((np.hypot(xx - wx, yy - wy) <= 2) & ~(np.isfinite(d) & ((flg & BAD) == 0))).sum())
    X, Y, D = xx[good], yy[good], d[good]
    rec = {"file": os.path.basename(path), "det": det, "mjd": round(mjd, 3), "epoch": epoch_of(mjd),
           "wave_um": round(float(wave[int(round(cy)), int(round(cx))]), 3), "fwhm_as": round(fwhm_as, 2),
           "n_bad_box": nbad, "bad_within2_C": badC, "bad_within2_W": badW, "sky_sigma_mjy_px": round(float(ssig) * to_mjy, 3)}
    if len(D) < 60:
        rec["status"] = "too_few_pixels"
        rows.append(rec)
        continue
    # (2) fixed positions C and W
    A = np.vstack([gauss(X, Y, cx, cy, sig), gauss(X, Y, wx, wy, sig), np.ones_like(X)]).T
    coef, *_ = np.linalg.lstsq(A, D, rcond=None)
    rec["fixC_mjy"] = round(float(coef[0]) * to_mjy, 2)
    rec["fixW_mjy"] = round(float(coef[1]) * to_mjy, 2)
    # (1) free single source (start at brightest good pixel within 3 px of C)
    near = good & (np.hypot(xx - cx, yy - cy) <= 3)
    if near.sum() == 0:
        rec["status"] = "no_good_pixels_near_C"
        rows.append(rec)
        continue
    i = np.argmax(np.where(near, d, -np.inf))
    x0, y0 = xx.flat[i], yy.flat[i]

    def res(p):
        return p[2] * gauss(X, Y, p[0], p[1], sig) + p[3] - D

    try:
        sol = least_squares(res, [x0, y0, max(d.flat[i], 1) * 2 * np.pi * sig ** 2, 0.0])
        fx, fy, famp = sol.x[0], sol.x[1], sol.x[2]
        fra, fdec = wcs.all_pix2world(fx, fy, 0)
        fc = SkyCoord(float(fra), float(fdec), unit="deg")
        rec.update(free_ra=round(float(fra), 5), free_dec=round(float(fdec), 5), free_mjy=round(float(famp) * to_mjy, 2),
                   free_off_C_as=round(fc.separation(SkyCoord(*C, unit="deg")).arcsec, 2),
                   free_off_W_as=round(fc.separation(SkyCoord(*W, unit="deg")).arcsec, 2),
                   free_pa_from_C=round(SkyCoord(*C, unit="deg").position_angle(fc).deg, 1), status="ok")
    except Exception as e:
        rec["status"] = f"fitfail {e}"
    rows.append(rec)

with open(os.path.join(OUT, "spherex_positions.json"), "w") as f:
    json.dump(rows, f, indent=1)

import collections
g = collections.defaultdict(list)
for r_ in rows:
    if r_.get("status") == "ok":
        g[(r_["epoch"], r_["det"])].append(r_)
print("epoch       det  n  median_free_off_C  median_free_off_W  med_free_mjy  med_fixC  med_fixW   (clean: no bad px within 2 of C and W)")
for k in sorted(g):
    L = [x for x in g[k] if x["bad_within2_C"] == 0 and x["bad_within2_W"] == 0]
    if not L:
        L2 = g[k]
        print(f"{k[0]} D{k[1]} {len(L2):2d}  (all have bad px near C or W)  fixC {np.median([x['fixC_mjy'] for x in L2]):8.1f} fixW {np.median([x['fixW_mjy'] for x in L2]):8.1f}")
        continue
    m = lambda key: np.median([x[key] for x in L])
    print(f"{k[0]} D{k[1]} {len(L):2d}  {m('free_off_C_as'):6.2f}  {m('free_off_W_as'):6.2f}  {m('free_mjy'):9.1f}  {m('fixC_mjy'):9.1f}  {m('fixW_mjy'):9.1f}   med RA,Dec {m('free_ra'):.5f} {m('free_dec'):.5f}")
