"""Zoomed crops (25x25 native px) around candidate A from cached cutouts; red = saturated rule,
blue = no value; green circle = 2.5 px aperture at the candidate; per-panel own stretch."""
import json, sys
from pathlib import Path
import numpy as np
from astropy.io import fits
from astropy.wcs import WCS
from PIL import Image, ImageDraw
sys.path.insert(0, ".")
from backend import photometry
RA, DEC = 270.80362, -23.46745
CACHE = Path("<repository>/backend/cache/sweep")
CUT = CACHE / "cutouts" / "250d2b7784"
pts = json.load(open(CACHE / "photometry" / "2c3c31c3c37a6a21.json"))["points"]
S = 10; H = 12
def crop(path, label):
    with fits.open(path) as h:
        img = h["IMAGE"].data.astype(float) - h["ZODI"].data.astype(float)
        flags = h["FLAGS"].data; w = WCS(h["IMAGE"].header)
    sat = photometry.saturated_pixels(flags, img)
    x, y = [float(v) for v in w.all_world2pix(RA, DEC, 0)]
    xi, yi = int(round(x)), int(round(y))
    sl = (slice(yi - H, yi + H + 1), slice(xi - H, xi + H + 1))
    c, s, fin = img[sl], sat[sl], np.isfinite(img[sl])
    lo, hi = np.nanpercentile(c, 10), np.nanmax(c)
    a = np.clip((np.nan_to_num(c, nan=hi) - lo) / (hi - lo + 1e-9), 0, 1) ** 0.5
    rgb = np.stack([a] * 3, -1) * 255
    rgb[s] = rgb[s] * 0.4 + np.array([255, 0, 0]) * 0.6
    rgb[~fin] = [40, 80, 255]
    im = Image.fromarray(rgb[::-1].astype("uint8")).resize((c.shape[1] * S, c.shape[0] * S), Image.NEAREST)
    d = ImageDraw.Draw(im)
    cx = (x - (xi - H) + 0.5) * S; cy = (c.shape[0] - (y - (yi - H)) - 0.5) * S
    r = 2.5 * S
    d.ellipse([cx - r, cy - r, cx + r, cy + r], outline=(0, 255, 0))
    d.text((3, 3), label, fill=(255, 255, 0))
    # print pixel of max within 3 px of candidate and nearest saturated cluster
    return im
out = Path(sys.argv[1]); dets = [int(v) for v in sys.argv[2].split(",")]
rows = []
for det in dets:
    for ep in (1, 2, 3):
        sel = [p for p in pts if p["detector"] == det and p["epoch"] == ep and p["status"] != "off_image"][:6]
        ims = [crop(CUT / f"{p['obs_id']}_D{det}.fits", f"D{det} {p['date'][:10]} {p['wavelength_um']:.2f} {p['status'][:3]}") for p in sel]
        Wd = sum(i.width + 4 for i in ims); row = Image.new("RGB", (Wd, ims[0].height), (0, 0, 0)); xo = 0
        for i in ims: row.paste(i, (xo, 0)); xo += i.width + 4
        rows.append(row)
sheet = Image.new("RGB", (max(r.width for r in rows), sum(r.height + 4 for r in rows)))
yo = 0
for r in rows: sheet.paste(r, (0, yo)); yo += r.height + 4
sheet.save(out)
