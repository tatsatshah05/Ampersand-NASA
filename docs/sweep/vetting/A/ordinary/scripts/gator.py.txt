"""IRSA Gator cone query (plain HTTP, IPAC table). Usage: gator.py CATALOG RA DEC RADIUS_ARCSEC OUTFILE [selcols] [constraints]"""
import sys, urllib.parse, urllib.request, time
cat, ra, dec, rad, out = sys.argv[1:6]
p = {"catalog": cat, "spatial": "cone", "objstr": f"{ra} {dec}", "radius": rad, "radunits": "arcsec", "outfmt": "1"}
if len(sys.argv) > 6 and sys.argv[6]:
    p["selcols"] = sys.argv[6]
if len(sys.argv) > 7:
    p["constraints"] = sys.argv[7]
url = "https://irsa.ipac.caltech.edu/cgi-bin/Gator/nph-query?" + urllib.parse.urlencode(p)
for k in range(6):
    try:
        data = urllib.request.urlopen(urllib.request.Request(url, headers={"User-Agent": "generic-python-client/1.0"}), timeout=240).read().decode()
        break
    except Exception as e:
        print("retry", k, e, file=sys.stderr); time.sleep(5 + 5 * k)
else:
    raise SystemExit("failed: " + url)
open(out, "w").write("\\ URL " + url + "\n" + data)
lines = [l for l in data.splitlines() if not l.startswith("\\")]
print("\n".join(lines[:60]))
